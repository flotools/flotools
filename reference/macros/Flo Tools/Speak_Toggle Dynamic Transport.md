# Speak/Toggle Dynamic Transport

- Group: Flo Tools
- UID: 276A1B62-57BA-49E9-AB62-A5F6324A48A7
- Enabled: True
- Triggers: Ctrl+Cmd+P

## Actions

- ExecuteJavaScriptForAutomation
  ```javascript
  currentCommand = "cCp"
  
  function single() {
  vo = Application("VoiceOver")
  sys = Application("System Events")
  pt = sys.applicationProcesses.whose({name:{_contains:"Pro Tools"}})[0]
  value = pt.menuBars[0].menuBarItems.byName("Options").menus.byName("Options").menuItems.byName("Dynamic Transport").attributes["AXMenuItemMarkChar"].value()
  if (value == null) vo.output ("Dynamic Transport")
  else vo.output("Dynamic Transport, Checked")
  }
  
  function double() {
  vo = Application("VoiceOver")
  sys = Application("System Events")
  pt = sys.applicationProcesses.whose({name:{_contains:"Pro Tools"}})[0]
  item = pt.menuBars[0].menuBarItems.byName("Options").menus.byName("Options").menuItems.byName("Dynamic Transport")
  item.click()
  }
  
  kme = Application("Keyboard Maestro Engine")
  command = kme.getvariable("command").split(".")
  start = new Date().getTime()
  kme.setvariable("command", {to:currentCommand+"."+start})
  if (currentCommand == command[0] & start-command[1]<kme.getvariable("FTTimeout")) {
  kme.playSound(Path("/System/Library/Sounds/Tink.aiff"))
  double()
  } else  {
  runSingle = true
  while (new Date().getTime()-start<kme.getvariable("FTTimeout")) {
  command = kme.getvariable("command").split(".")
  if (currentCommand == command[0] & command[1]>start) {
  runSingle=false
  break;
  }
  }
  if (runSingle) single()
  }
  ```
