  currentCommand = "c"
  
  function single() {
  vo = Application("VoiceOver")
  sys = Application("System Events")
  pt = sys.applicationProcesses.whose({name:{_contains:"Pro Tools"}})[0]
  windows = pt.windows
  mix = windows.whose({name:{_beginsWith:"Mix:"}})[0]
  track = mix.groups.byName(vo.voCursor.textUnderCursor())
  field = track.textFields.whose({name:{_beginsWith:"Comments"}})[0]
  value = field.value()
  if (value == "") vo.output("No Comment")
  vo.output(value)
  }

  vo = Application("VoiceOver")

  mix = windows.whose({name:{_beginsWith:"Mix:"}})[0]
  track = mix.groups.byName(vo.voCursor.textUnderCursor())
  field = track.textFields.whose({name:{_beginsWith:"Comments"}})[0]
  field.click()

  if (currentCommand == command[0] & start-command[1]<kme.getvariable("FTTimeout")) {
  double()
